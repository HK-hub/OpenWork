### 2.2 数据库迁移系统

```rust
/// 迁移管理器
pub struct MigrationManager {
    db_path: PathBuf,
    migrations: Vec<Migration>,
}

pub struct Migration {
    pub version: i32,
    pub description: String,
    pub up: String,     // 升级 SQL
    pub down: String,   // 回滚 SQL
}

impl MigrationManager {
    /// 注册所有迁移
    pub fn new(db_path: PathBuf) -> Self {
        let migrations = vec![
            Migration {
                version: 1,
                description: "Initial schema v2.0".into(),
                up: include_str!("../migrations/001_initial_schema.sql").into(),
                down: "DROP TABLE IF EXISTS ...".into(), // 简略
            },
            // 后续迁移在此注册...
        ];
        Self { db_path, migrations }
    }

    /// 执行迁移 (带完整事务回滚)
    pub async fn migrate(&self, conn: &Connection) -> Result<(), DbError> {
        // 1. 获取当前 schema 版本
        let current_version: i32 = conn
            .query_row(
                "SELECT COALESCE(MAX(version), 0) FROM schema_version",
                [],
                |row| row.get(0),
            )
            .unwrap_or(0);

        // 2. 筛选未应用的迁移
        let pending: Vec<&Migration> = self.migrations
            .iter()
            .filter(|m| m.version > current_version)
            .collect();

        if pending.is_empty() {
            log::info!("Database schema is up-to-date (version {})", current_version);
            return Ok(());
        }

        // 3. 迁移前自动备份
        self.backup_before_migration(pending.last().unwrap().version)?;

        // 4. 所有迁移在同一个事务中执行 (全部成功或全部回滚)
        conn.execute("BEGIN IMMEDIATE")?;
        
        let result = (|| -> Result<(), DbError> {
            for migration in &pending {
                log::info!("Applying migration {}: {}", migration.version, migration.description);
                
                // 执行 up SQL
                conn.execute_batch(&migration.up)?;

                // 记录版本
                conn.execute(
                    "INSERT INTO schema_version (version, description) VALUES (?1, ?2)",
                    params![migration.version, migration.description],
                )?;
            }
            Ok(())
        })();

        match result {
            Ok(()) => {
                conn.execute("COMMIT")?;
                log::info!("All {} migration(s) applied successfully", pending.len());
                Ok(())
            }
            Err(e) => {
                // 事务回滚: 所有已执行的迁移全部撤销
                conn.execute("ROLLBACK")?;
                log::error!("Migration failed at version {}: {:?}. All changes rolled back.", 
                    pending.first().map(|m| m.version).unwrap_or(0), e);
                
                // 恢复备份
                self.restore_from_backup()?;
                
                Err(DbError::MigrationFailed(format!(
                    "Migration rolled back. Database restored from backup. Error: {}", e
                )))
            }
        }
    }
}
```

```rust
impl LoopEngineImpl {
    /// 主循环入口
    pub async fn run(&self, input: TaskInput) -> Result<SessionResult, LoopError> {
        // 1. 触发 on_session_start hooks
        self.hooks_runtime.trigger(HookTrigger::OnSessionStart, &ctx).await?;

        // 2. 初始化循环状态
        self.initialize_loop(&input).await?;

        // 3. 主循环
        loop {
            // 检查取消信号
            if self.cancel_token.is_cancelled() {
                return self.handle_cancellation().await;
            }

            // 检查终止条件
            if let Some(reason) = self.check_termination().await {
                return self.finalize(reason).await;
            }

            // 增加迭代计数
            self.state.write().await.iteration += 1;
            self.state.write().await.current_step += 1;

            // === Phase 1: Perceive ===
            self.emit_event(AgentEvent::phase_start("perceive")).await;
            let context = self.perceive(&input).await?;
            self.emit_event(AgentEvent::phase_end("perceive")).await;

            // === Phase 2: Act ===
            self.emit_event(AgentEvent::phase_start("act")).await;
            let action = self.act(&context).await?;
            self.emit_event(AgentEvent::phase_end("act")).await;

            // === Phase 3: Observe ===
            self.emit_event(AgentEvent::phase_start("observe")).await;
            let observation = self.observe(&action).await?;
            self.emit_event(AgentEvent::phase_end("observe")).await;

            // === Phase 4: Reflect (条件触发) ===
            if self.state.read().await.iteration % self.config.loop_config.reflection_interval == 0 {
                self.emit_event(AgentEvent::phase_start("reflect")).await;
                self.reflect(&observation).await?;
                self.emit_event(AgentEvent::phase_end("reflect")).await;
            }

            // 保存 Loop Step
            self.persist_loop_step().await?;

            // 自动创建检查点
            if self.should_checkpoint() {
                self.create_checkpoint().await?;
            }
        }
    }

    /// Perceive 阶段
    async fn perceive(&self, input: &TaskInput) -> Result<AssembledContext, LoopError> {
        // 1. 从 Context Engine 获取组装后的上下文
        let context = self.context_engine.assemble(
            &self.session.id,
            input,
            &self.config.context_config,
        ).await?;

        // 2. 从 Prompt Engine 获取组装后的 Prompt
        let prompt = self.prompt_engine.assemble(
            &context,
            &self.state.read().await.message_history,
            &self.config.prompt_config,
        ).await?;

        // 3. 构建完整的 messages 数组
        let messages = self.build_messages(&prompt, &context);

        // 4. 持久化 prompt 快照
        self.save_prompt_snapshot(&prompt).await?;

        Ok(AssembledContext { context, prompt, messages })
    }

    /// Act 阶段
    /// 
    /// 容错策略: LLM 响应不可靠，需要多层解析 + 重试 + 降级
    async fn act(&self, ctx: &AssembledContext) -> Result<ActionResult, LoopError> {
        // 1. 触发 pre_model_call hooks
        let messages = self.hooks_runtime
            .trigger_intercept(HookTrigger::PreModelCall, &ctx.messages)
            .await?;

        // 2. 模型推理
        let response = self.model_router
            .chat_stream(messages, &self.config.model_config)
            .await?;

        // 3. 触发 post_model_call hooks
        let response = self.hooks_runtime
            .trigger_intercept(HookTrigger::PostModelCall, &response)
            .await?;

        // 4. 解析 LLM 响应 (多层容错 + 重试)
        self.parse_and_handle_response(response).await
    }

    /// 解析 LLM 响应并处理 (多层容错)
    async fn parse_and_handle_response(
        &self,
        response: ChatResponse,
    ) -> Result<ActionResult, LoopError> {
        match response {
            ChatResponse::Text(text) => {
                self.append_assistant_message(text.clone()).await?;
                Ok(ActionResult::TextResponse(text))
            }
            ChatResponse::ToolCalls(tool_calls) => {
                // 验证每个工具调用: 名称 + 参数
                let validated = self.validate_tool_calls(tool_calls).await?;
                let results = self.execute_tools(validated).await?;
                Ok(ActionResult::ToolResults(results))
            }
            ChatResponse::Mixed { text, tool_calls } => {
                // GPT-4 常见: 部分文本 + 部分工具调用
                self.append_assistant_message(text.clone()).await?;
                let validated = self.validate_tool_calls(tool_calls).await?;
                let results = self.execute_tools(validated).await?;
                Ok(ActionResult::MixedResponse { text, results })
            }
        }
    }

    /// 验证工具调用 (名称存在性 + 参数 Schema 校验)
    async fn validate_tool_calls(
        &self,
        tool_calls: Vec<ToolCallRequest>,
    ) -> Result<Vec<ToolCallRequest>, LoopError> {
        let mut validated = Vec::new();
        for tc in tool_calls {
            // 工具名称检查
            let tool = self.tool_registry
                .get(&tc.tool_name)
                .await
                .ok_or_else(|| {
                    let available = self.tool_registry.list_names();
                    LoopError::ToolNotFound(format!(
                        "Tool '{}' not found. Available: {:?}", tc.tool_name, available
                    ))
                })?;
            // 参数 JSON Schema 校验
            tool.validate_args(&tc.args)
                .map_err(|e| LoopError::InvalidToolArgs(
                    tc.tool_name.clone(), e.to_string()
                ))?;
            validated.push(tc);
        }
        Ok(validated)
    }

    /// 执行工具调用
    async fn execute_tools(
        &self,
        tool_calls: Vec<ToolCallRequest>,
    ) -> Result<Vec<ToolResult>, LoopError> {
        let mut results = Vec::new();

        // 并发执行独立工具调用
        let futures: Vec<_> = tool_calls.into_iter().map(|tc| {
            self.execute_single_tool(tc)
        }).collect();

        for result_future in futures {
            match result_future.await {
                Ok(result) => {
                    // 持久化工具调用
                    self.save_tool_call(&result).await?;
                    results.push(result);
                }
                Err(e) => {
                    // 工具错误 → 返回结构化错误给 LLM
                    let error_result = ToolResult::error(&tc.id, &e);
                    self.save_tool_call(&error_result).await?;
                    results.push(error_result);
                }
            }
        }

        Ok(results)
    }

    /// 执行单个工具 (含权限检查)
    async fn execute_single_tool(
        &self,
        tc: ToolCallRequest,
    ) -> Result<ToolResult, LoopError> {
        // 1. 触发 pre_act hooks
        let tc = self.hooks_runtime
            .trigger_intercept(HookTrigger::PreAct, &tc)
            .await?;

        // 2. 权限检查
        let decision = self.permission_gateway.check(
            &tc.tool_name,
            &tc.args,
            &PermissionContext::from_session(&self.session),
        ).await?;

        match decision {
            PermissionDecision::Denied => {
                return Ok(ToolResult::denied(&tc.id, &decision.reason));
            }
            PermissionDecision::RequiresApproval => {
                // 发送确认请求到 UI
                let approved = self.request_user_approval(&tc).await?;
                if !approved {
                    return Ok(ToolResult::denied(&tc.id, "User denied"));
                }
            }
            PermissionDecision::Allowed => {
                // 继续执行
            }
        }

        // 3. 在沙箱中执行工具
        let tool = self.tool_registry.get(&tc.tool_name).await
            .ok_or(LoopError::ToolNotFound(tc.tool_name.clone()))?;

        let result = self.sandbox.execute(|| {
            tool.invoke(tc.args.clone(), &self.tool_context())
        }).await?;

        // 4. 记录审计日志
        self.audit_log(AuditAction::ToolCall {
            tool_name: tc.tool_name.clone(),
            args_summary: summarize(&tc.args),
            result_summary: summarize(&result),
            approved: true,
        }).await?;

        // 5. 触发 post_observe / on_tool_result hooks
        let result = self.hooks_runtime
            .trigger_intercept(HookTrigger::PostObserve, &result)
            .await?;

        Ok(result)
    }

    /// Observe 阶段
    async fn observe(&self, action: &ActionResult) -> Result<Observation, LoopError> {
        match action {
            ActionResult::TextResponse(text) => {
                Ok(Observation::text(text))
            }
            ActionResult::ToolResults(results) => {
                // 解析工具结果为结构化 Observation
                let observations = results.iter().map(|r| {
                    Observation::from_tool_result(r)
                }).collect();

                // 写入工作记忆
                self.state.write().await.working_memory
                    .insert("last_observation".into(), json!(observations));

                Ok(Observation::tool_results(observations))
            }
        }
    }

    /// Reflect 阶段
    async fn reflect(&self, observation: &Observation) -> Result<(), LoopError> {
        // 1. 构建反思 Prompt
        let reflection_prompt = format!(
            "评估当前任务进度。已完成: {:?}。当前观察: {:?}。\
             是否需要调整计划？是否需要不同的策略？",
            self.state.read().await.working_memory,
            observation,
        );

        // 2. LLM 自评
        let reflection = self.model_router
            .chat_simple(&reflection_prompt)
            .await?;

        // 3. 判断是否需要重规划
        if reflection.contains("调整计划") || reflection.contains("偏离") {
            self.trigger_replanning().await?;
        }

        // 4. 更新工作记忆
        self.state.write().await.working_memory
            .insert("last_reflection".into(), json!(reflection));

        Ok(())
    }
}
```

```rust
    pub fn build_l2(&self, ctx: &AssemblyContext) -> Result<LayerContent, PromptError> {
        let persona = ctx.profile.as_ref()
            .map(|p| p.persona.as_str())
            .unwrap_or("concise");

        let template = self.get_persona_template(persona)?;

        Ok(LayerContent {
            layer: PromptLayer::L2Persona,
            content: template,
            source: "profile",
            can_override: true,
        })
    }

    /// 构建 L3: Capabilities (动态生成工具清单)
    pub fn build_l3(&self, ctx: &AssemblyContext) -> Result<LayerContent, PromptError> {
        let tools = ctx.available_tools;
        let mut content = String::from("## 可用工具\n\n");

        for tool in tools {
            content.push_str(&format!(
                "- **{}**: {} (风险等级: {}, 需确认: {})\n",
                tool.name, tool.description, tool.risk_level, tool.requires_approval,
            ));
            if let Some(schema) = &tool.parameters_schema {
                content.push_str(&format!("  参数: {}\n", schema));
            }
        }

        // 添加 MCP 工具
        if !ctx.mcp_tools.is_empty() {
            content.push_str("\n### MCP 工具\n\n");
            for (server, tools) in &ctx.mcp_tools {
                content.push_str(&format!("**{}**:\n", server));
                for tool in tools {
                    content.push_str(&format!("  - {}\n", tool));
                }
            }
        }

        // 添加 Skills
        if !ctx.active_skills.is_empty() {
            content.push_str("\n### 可用 Skills\n\n");
            for skill in &ctx.active_skills {
                content.push_str(&format!("- **{}**: {}\n", skill.name, skill.description));
            }
        }

        Ok(LayerContent {
            layer: PromptLayer::L3Capabilities,
            content,
            source: "dynamic",
            can_override: false, // 能力声明不可覆盖
        })
    }

    /// 合并所有层级
    pub fn merge(&self, layers: Vec<LayerContent>) -> MergedPrompt {
        // 按层级排序
        let mut sorted = layers;
        sorted.sort_by_key(|l| l.layer);

        // 合并: 后者覆盖前者 (对于可覆盖的层)
        let mut sections = Vec::new();
        let mut seen_override = HashSet::new();

        for layer in sorted.iter().rev() {
            if layer.can_override && !seen_override.insert(layer.layer) {
                continue; // 已被更高层级覆盖
            }
            sections.push(layer.clone());
        }

        sections.reverse(); // 恢复 L1 → L7 顺序

        let total_tokens: u32 = sections.iter()
            .map(|s| estimate_tokens(&s.content))
            .sum();

        MergedPrompt {
            sections,
            total_tokens,
        }
    }
```

#### 3.2.3 Token 预算分配策略

> 问题: 七层模型的总 Token 必须控制在模型窗口内。当 L6 (Dynamic Context) 膨胀时，如何分配预算？

**各层 Token 预算分配 (以 100K 模型窗口为例)**:

| 层级 | 预算 (tokens) | 占比 | 优先级 | 弹性范围 | 说明 |
|------|:---:|:---:|:---:|:---:|------|
| L1 Base Identity | ~500 | 0.5% | 最高 (不可压缩) | 固定 | 系统身份定义 |
| L2 Persona | ~300 | 0.3% | 高 (不可压缩) | 固定 | 风格指令 |
| L3 Capabilities | ~2,000 | 2% | 高 (仅截断) | 1K-3K | 工具/Skills 清单 |
| L4 Constraints | ~1,500 | 1.5% | 高 (仅截断) | 1K-3K | 安全约束 + 项目规则 |
| L5 Task Template | ~500 | 0.5% | 中 (可压缩) | 200-1K | 任务类型模板 |
| L6 Dynamic Context | ~40,000 | 40% | 中 (压缩) | 10K-60K | 文件/Git/检索/历史 |
| L7 User Override | ~1,000 | 1% | 最高 (不可压缩) | 0-5K | 用户显式注入 |
| **系统保留** | **~54,200** | **54%** | — | — | — |
| **输出预留** | ~16,000 | 16% | — | — | LLM 生成输出 |
| **工具预留** | ~8,000 | 8% | — | — | 工具调用 + 结果 |
| **动态缓冲** | ~21,800 | 22% | — | — | 流式响应缓冲 |

**压缩优先级** (预算不足时):
1. **L6 Dynamic Context** → 首先压缩 (分层摘要/尾部截断)
2. **L3 Capabilities** → 截断不常用工具描述 (保留前 10 个高频工具)
3. **L5 Task Template** → 精简模板
4. **L4 Constraints** → 合并重复约束
5. **L1/L2/L7** → 从不压缩 (身份/风格/用户指令不可丢)

```rust
pub struct TokenBudgetAllocator {
    model_window: usize,      // 模型总窗口 (如 100000)
    output_reserve: usize,    // 输出预留 (如 16000)
    tool_reserve: usize,      // 工具预留 (如 8000)
    dynamic_buffer: usize,    // 动态缓冲 (如 21800)
}

impl TokenBudgetAllocator {
    /// 计算各层可用 token
    pub fn allocate(&self, ctx: &AssemblyContext) -> HashMap<PromptLayer, usize> {
        let available = self.model_window 
            - self.output_reserve 
            - self.tool_reserve 
            - self.dynamic_buffer;
        
        // 固定分配
        let mut budget = HashMap::new();
        budget.insert(PromptLayer::L1BaseIdentity, 500.min(available));
        budget.insert(PromptLayer::L2Persona, 300.min(available));
        budget.insert(PromptLayer::L7UserOverride, ctx.user_override_len().min(available));
        
        // 剩余按比例分配
        let used: usize = budget.values().sum();
        let remaining = available.saturating_sub(used);
        
        budget.insert(PromptLayer::L6DynamicContext, (remaining as f64 * 0.70) as usize);
        budget.insert(PromptLayer::L3Capabilities, (remaining as f64 * 0.15) as usize);
        budget.insert(PromptLayer::L4Constraints, (remaining as f64 * 0.10) as usize);
        budget.insert(PromptLayer::L5TaskTemplate, (remaining as f64 * 0.05) as usize);
        
        budget
    }
}
```

```rust
/// 上下文组装器
pub struct ContextAssembler {
    collectors: Vec<Box<dyn ContextCollector>>,
    retriever: Arc<dyn HybridRetriever>,
    compressor: Arc<dyn ContextCompressor>,
    memory_manager: Arc<dyn MemoryManager>,
    config: ContextConfig,
}

#[async_trait]
pub trait ContextCollector: Send + Sync {
    /// 采集器名称
    fn name(&self) -> &str;

    /// 来源类型
    fn source_type(&self) -> ContextSourceType;

    /// 默认权重
    fn default_weight(&self) -> f64;

    /// 采集上下文
    async fn collect(&self, ctx: &CollectionContext) -> Result<Vec<ContextItem>, ContextError>;
}

impl ContextAssembler {
    /// 组装完整上下文
    pub async fn assemble(
        &self,
        session_id: &str,
        task_input: &TaskInput,
        config: &ContextConfig,
    ) -> Result<AssembledContext, ContextError> {
        let mut all_items = Vec::new();

        // 1. 从所有采集器收集上下文
        for collector in &self.collectors {
            if !config.is_source_enabled(collector.source_type()) {
                continue;
            }

            let weight = config.get_source_weight(collector.source_type())
                .unwrap_or(collector.default_weight());

            let items = collector.collect(&CollectionContext {
                session_id: session_id.to_string(),
                task_input: task_input.clone(),
            }).await?;

            // 应用权重
            for item in items {
                all_items.push(ContextItem {
                    weight: weight * item.weight,
                    ..item
                });
            }
        }

        // 2. 添加检索结果 (RAG)
        let query = self.build_retrieval_query(task_input, &all_items);
        let retrieved = self.retriever.search(&query, config.retrieval_top_k).await?;
        for item in retrieved {
            all_items.push(ContextItem {
                source: ContextSourceType::Retrieval,
                weight: config.retrieval_weight,
                ..item
            });
        }

        // 3. 添加记忆
        let memories = self.memory_manager
            .retrieve_relevant(&query, config.memory_top_k)
            .await?;
        for mem in memories {
            all_items.push(ContextItem {
                source: ContextSourceType::Memory,
                weight: config.memory_weight,
                content: mem.content,
                metadata: Some(json!({ "memory_id": mem.id, "category": mem.category })),
                ..Default::default()
            });
        }

        // 4. 按权重排序
        all_items.sort_by(|a, b| b.weight.partial_cmp(&a.weight).unwrap_or(Ordering::Equal));

        // 5. 应用 Token 预算: 压缩超限内容
        let budget = self.calculate_token_budget(config);
        let assembled = self.compressor.compress(all_items, budget, &config.compression).await?;

        Ok(assembled)
    }
}
```

```rust
        match strategy {
            CompressionStrategy::CompressDynamic { recency_window, summary_chunk, extract_facts } => {
                self.compress_dynamic(items, budget, *recency_window, *summary_chunk, *extract_facts).await
            }
            CompressionStrategy::TruncateTail { max_tokens } => {
                self.truncate_tail(items, *max_tokens)
            }
            CompressionStrategy::SummarizeAll => {
                self.summarize_all(items, budget).await
            }
            CompressionStrategy::DropLowWeight { keep_ratio } => {
                self.drop_low_weight(items, *keep_ratio)
            }
        }
    }

    /// 动态压缩: 近因 + 摘要 + 事实
    async fn compress_dynamic(
        &self,
        items: Vec<ContextItem>,
        budget: TokenBudget,
        recency_window: usize,
        summary_chunk: usize,
        extract_facts: bool,
    ) -> Result<AssembledContext, ContextError> {
        let mut result = AssembledContext::new();

        // 分离近期和远期
        let (recent, older) = items.split_at(recency_window.min(items.len()));

        // 近期: 原样保留
        for item in recent {
            result.add_item(item.clone());
        }

        // 剩余 token 预算
        let remaining = budget.remaining(result.total_tokens());

        // 远期: 分块摘要
        if !older.is_empty() && remaining > 0 {
            let chunks: Vec<&[ContextItem]> = older.chunks(summary_chunk).collect();

            for chunk in chunks {
                let chunk_text: String = chunk.iter()
                    .map(|i| i.content.clone())
                    .collect::<Vec<_>>()
                    .join("\n---\n");

                let summary = self.model.summarize(
                    &format!("请用 1-2 句话摘要以下内容，保留关键信息:\n\n{}", chunk_text),
                    remaining / chunks.len() as u32,
                ).await?;

                result.add_summary(summary);

                // 抽取结构化事实
                if extract_facts {
                    let facts = self.extract_facts(&chunk_text).await?;
                    for fact in facts {
                        result.add_fact(fact);
                    }
                }
            }
        }

        Ok(result)
    }

    /// 从上下文抽取结构化事实
    async fn extract_facts(&self, text: &str) -> Result<Vec<Fact>, ContextError> {
        let prompt = format!(
            "从以下内容中提取关键事实、决策和约束。\
             每个事实格式: [类型]: [内容]\n\
             类型可选: decision(决策), constraint(约束), finding(发现), todo(待办)\n\n{}",
            text
        );

        let response = self.model.chat_simple(&prompt).await?;

        // 解析事实
        let facts = response.lines()
            .filter_map(|line| {
                let parts: Vec<&str> = line.splitn(2, ": ").collect();
                if parts.len() == 2 {
                    Some(Fact {
                        category: parts[0].trim().to_string(),
                        content: parts[1].trim().to_string(),
                    })
                } else {
                    None
                }
            })
            .collect();

        Ok(facts)
    }
}
```

```rust
            .ok_or(ToolError::InvalidArgs("path is required"))?

            .ok_or(ToolError::InvalidArgs("edits array is required"))?;

        ctx.sandbox.validate_path_write(&full_path)?;

        let mut content = tokio::fs::read_to_string(&full_path).await
            .map_err(|e| ToolError::Execution(format!("读取文件失败: {}", e)))?;

        let mut changes = Vec::new();
        let original_content = content.clone();

        for (i, edit) in edits.iter().enumerate() {
            let old_str = edit["old_string"].as_str()
                .ok_or(ToolError::InvalidArgs(format!("edits[{}].old_string required", i)))?;
            let new_str = edit["new_string"].as_str()
                .ok_or(ToolError::InvalidArgs(format!("edits[{}].new_string required", i)))?;
            let replace_all = edit["replace_all"].as_bool().unwrap_or(false);

                let count = content.matches(old_str).count();
                content = content.replace(old_str, new_str);
                changes.push(EditChange {
                    index: i,
                    old: old_str.to_string(),
                    new: new_str.to_string(),

                    index: i,
                    old: old_str.to_string(),
                    new: new_str.to_string(),
                    occurrences: 1,
                });

            .map_err(|e| ToolError::Execution(format!("写入文件失败: {}", e)))?;

            tool_name: self.meta().name.clone(),

            content: json!({
                "path": path,
                "changes_count": changes.len(),
                "changes": changes,
                "diff": diff,

        })
```

```rust
pub struct PermissionGatewayImpl {
    config: PermissionConfig,
    approval_cache: RwLock<HashMap<String, ApprovalEntry>>,

#[derive(Debug, Clone)]
pub struct ApprovalEntry {
    pub tool_name: String,
    pub level: ApprovalLevel,
    pub granted_at: i64,
    pub expires_at: Option<i64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApprovalLevel {
    Once,
    Session(String),
    Project(String),
    Permanent,
}

impl PermissionGatewayImpl {
    /// 检查权限
    pub async fn check(
        &self,
        tool_name: &str,
        args: &Value,
        context: &PermissionContext,
    ) -> Result<PermissionDecision, PermissionError> {
        let tool = self.get_tool_meta(tool_name)?;

        // 1. 风险等级判断
        match tool.risk_level {
            RiskLevel::L0 => {
                return Ok(PermissionDecision::Allowed);
            }
            RiskLevel::L3 => {
                // 永远需要确认 + 不可擦除审计
                self.audit_logger.log_critical(tool_name, args, context).await?;
                return Ok(PermissionDecision::RequiresApproval {
                    reason: "关键操作，需要人工确认".into(),
                    dry_run: self.generate_dry_run(tool_name, args).await?,
                });
            }
            _ => {}
        }

        // 2. 检查 per-tool 策略
        if let Some(policy) = self.config.per_tool.get(tool_name) {
            match policy {
                ToolPolicy::Auto => return Ok(PermissionDecision::Allowed),
                ToolPolicy::Deny => return Ok(PermissionDecision::Denied {
                    reason: format!("工具 '{}' 已被管理员禁用", tool_name),
                }),
                ToolPolicy::AlwaysConfirm => {
                    return Ok(PermissionDecision::RequiresApproval {
                        reason: format!("工具 '{}' 需要每次确认", tool_name),
                        dry_run: self.generate_dry_run(tool_name, args).await?,
                    });
                }
                ToolPolicy::Approve => {} // 继续检查缓存
            }
        }

        // 3. 检查审批缓存
        let cache = self.approval_cache.read().await;
        if let Some(entry) = cache.get(tool_name) {
            match &entry.level {
                ApprovalLevel::Session(sid) if sid == &context.session_id => {
                    return Ok(PermissionDecision::Allowed);
                }
                ApprovalLevel::Project(pid) if pid == &context.project_id => {
                    return Ok(PermissionDecision::Allowed);
                }
                ApprovalLevel::Permanent => {
                    return Ok(PermissionDecision::Allowed);
                }
                _ => {}
            }
        }

        // 4. 默认策略: 需要确认
        Ok(PermissionDecision::RequiresApproval {
            reason: format!("工具 '{}' 需要确认 (风险等级: {:?})", tool_name, tool.risk_level),
            dry_run: self.generate_dry_run(tool_name, args).await?,
        })
    }

    /// 记录批准
    pub async fn record_approval(
        &self,
        tool_name: &str,
        level: ApprovalLevel,
    ) {
        let mut cache = self.approval_cache.write().await;
        cache.insert(tool_name.to_string(), ApprovalEntry {
            tool_name: tool_name.to_string(),
            level,
            granted_at: current_timestamp(),
            expires_at: None, // 可根据级别设置过期时间
        });
    }

    /// 生成 dry-run 预览
    async fn generate_dry_run(
        &self,
        tool_name: &str,
        args: &Value,
    ) -> Result<Option<String>, PermissionError> {
        match tool_name {
            "delete_file" => {
                let path = args["path"].as_str().unwrap_or("?");
                // 获取文件信息
                let info = self.get_file_info(path).await?;
                Ok(Some(format!(
                    "将删除文件: {}\n大小: {} 行\n最后修改: {}",
                    path, info.line_count, info.last_modified,
                )))
            }
            "run_command" => {
                let cmd = args["command"].as_str().unwrap_or("?");
                Ok(Some(format!("将执行命令: {}", cmd)))
            }
            "edit_file" => {
                let path = args["path"].as_str().unwrap_or("?");
                let edits = args["edits"].as_array().map(|a| a.len()).unwrap_or(0);
                Ok(Some(format!(
                    "将在文件 {} 中执行 {} 处编辑", path, edits
                )))
            }
            _ => Ok(None),
        }
    }
}
```

```rust
pub struct MCPClient {

impl MCPClient {

    pub async fn start_all(&self) -> Vec<Result<(), MCPError>> {

        for (id, config) in &self.server_configs {

                continue;
            }

    /// 启动单个 MCP Server
    async fn start_server(
        &self,
        id: &str,
        config: &MCPServerConfig,
    ) -> Result<(), MCPError> {
        let transport = match config.transport {
            TransportType::Stdio => {
                MCPTransport::stdio(
                    config.command.as_ref().ok_or(MCPError::MissingCommand)?,
                    &config.args,
                    &config.env,
                )
            }
            TransportType::SSE => {
                MCPTransport::sse(
                    config.url.as_ref().ok_or(MCPError::MissingUrl)?,
                    &config.headers,
                )
            }
            TransportType::HTTP => {
                MCPTransport::http(
                    config.url.as_ref().ok_or(MCPError::MissingUrl)?,
                    &config.headers,
                )
            }
        };

        let connection = MCPConnection::new(transport).await?;

        // 初始化
        let init_result = connection.initialize(InitializeRequest {
            protocol_version: "2024-11-05".into(),
            client_info: ClientInfo {
                name: "OpenCoding Agent Desktop".into(),
                version: env!("CARGO_PKG_VERSION").into(),
            },
        }).await?;

        // 发现工具
        let tools = connection.list_tools().await?;

        // 注册到 ToolRegistry
        for tool in &tools {
            self.tool_registry.register(Arc::new(MCPToolAdapter {
                server_id: id.to_string(),
                server_name: config.name.clone(),
                tool: tool.clone(),
                connection: connection.clone(),
            })).await?;
        }

        // 保存连接
        self.connections.write().await.insert(id.to_string(), Arc::new(connection));

        // 启动健康检查
        if config.auto_restart {
            self.start_health_check(id, config).await;
        }

        Ok(())
    }

    /// 健康检查 + 自动重连
    async fn start_health_check(&self, id: &str, config: &MCPServerConfig) {
        let id = id.to_string();
        let config = config.clone();
        let this = self.clone(); // 需要实现 Clone

        tokio::spawn(async move {
            loop {
                tokio::time::sleep(Duration::from_secs(config.health_check_interval_secs)).await;

                let connections = this.connections.read().await;
                if let Some(conn) = connections.get(&id) {
                    match conn.ping().await {
                        Ok(_) => {
                            // 健康
                            this.update_server_status(&id, "connected", None).await;
                        }
                        Err(e) => {
                            log::warn!("MCP Server '{}' health check failed: {:?}", config.name, e);
                            this.update_server_status(&id, "error", Some(&e.to_string())).await;

                            if config.auto_restart {
                                log::info!("Restarting MCP Server '{}'...", config.name);
                                drop(connections);
                                if let Err(e) = this.restart_server(&id).await {
                                    log::error!("Failed to restart: {:?}", e);
                                }
                            }
                        }
                    }
                }
            }
        });
    }
```

```rust
    ) -> Result<(), HookError> {
        let matching: Vec<&RegisteredHook> = self.hooks.iter()
            .filter(|h| h.config.trigger == trigger && h.config.enabled)
            .collect();

        for hook in matching {
            let result = self.execute_hook(hook, context).await;

                Ok(_) => {}
                Err(e) => {
                    match hook.config.on_error {
                        HookErrorPolicy::Skip => {
                            log::debug!("Hook '{}' error skipped: {:?}", hook.config.name, e);
                        }
                        HookErrorPolicy::Warn => {
                            log::warn!("Hook '{}' error: {:?}", hook.config.name, e);
                        }
                        HookErrorPolicy::Abort => {
                            return Err(HookError::HookAborted(hook.config.name.clone(), e.to_string()));
                        }
                    }
                }
            }
        }

        Ok(())
    }

    /// 触发拦截型 Hook (可修改输入/输出)
    pub async fn trigger_intercept<T: Serialize + DeserializeOwned>(
        &self,
        trigger: HookTrigger,
        data: &T,
    ) -> Result<T, HookError> {
        let matching: Vec<&RegisteredHook> = self.hooks.iter()
            .filter(|h| h.config.trigger == trigger && h.config.enabled && h.config.can_intercept)
            .collect();

        let mut current = serde_json::to_value(data)?;

        for hook in matching {
            let input = current.clone();
            let result = self.execute_hook_with_input(hook, &input).await?;

            if let Some(modified) = result.modified_data {
                current = modified;
            }

            if result.abort {
                return Err(HookError::Intercepted(hook.config.name.clone()));
            }
        }

        Ok(serde_json::from_value(current)?)
    }

    /// 在沙箱中执行 Hook 脚本
    async fn execute_hook(
        &self,
        hook: &RegisteredHook,
        context: &HookContext,
    ) -> Result<HookExecutionResult, HookError> {
        let input = json!({
            "hook_name": hook.config.name,
            "trigger": hook.config.trigger.to_string(),
            "context": context,
        });

        let output = self.sandbox.execute_script(
            &hook.config.script,
            &hook.config.language,
            &input,
            hook.config.timeout_secs,
        ).await?;

        Ok(serde_json::from_str(&output)?)
    }
```

```rust
                    self.merge_with_rename(&conn, table).await?;
                }
            }
        }

        conn.execute("DETACH DATABASE import_db", [])?;

        Ok(())
```

```rust
/// 索引条目
pub struct IndexEntry {

    pub file_path: String,
    pub symbol_name: String,
    pub symbol_kind: SymbolKind,  // function | class | variable | module | ...
    pub start_line: u32,

impl Indexer {
    /// 全量索引
    pub async fn full_index(&mut self) -> Result<(), IndexError> {
        self.status = IndexStatus::Indexing {
            progress: 0.0, files_done: 0, files_total: 0,
        };

        // 1. 收集所有需要索引的文件
        let files = self.collect_files()?;
        let total = files.len();
        self.status = IndexStatus::Indexing {
            progress: 0.0, files_done: 0, files_total: total,
        };

        // 2. 批量处理
        let mut batch = Vec::with_capacity(self.config.batch_size);
        for (i, file) in files.iter().enumerate() {
            match self.index_file(file).await {
                Ok(entries) => batch.extend(entries),
                Err(e) => log::warn!("Failed to index {}: {}", file.display(), e),
            }

            // 批量提交
            if batch.len() >= self.config.batch_size || i == total - 1 {
                self.db.batch_insert_index_entries(&batch).await?;
                batch.clear();
            }

            // 更新进度
            self.status = IndexStatus::Indexing {
                progress: (i + 1) as f32 / total as f32,
                files_done: i + 1,
                files_total: total,
            };
        }

        // 3. 记录索引元数据
        self.db.update_index_meta(&IndexMeta {
            project_path: self.config.project_root.to_string_lossy().into(),
            last_full_index: current_timestamp(),
            total_files: total,
            total_symbols: self.db.count_index_entries().await?,
            status: "complete".into(),
        }).await?;

        self.status = IndexStatus::Idle;
        Ok(())
    }

    /// 单文件索引
    async fn index_file(&self, file_path: &Path) -> Result<Vec<IndexEntry>, IndexError> {
        // 1. 检查文件是否需要重新索引 (hash 对比)
        let content = tokio::fs::read_to_string(file_path).await?;
        let hash = blake3::hash(content.as_bytes()).to_hex().to_string();
        
        if let Some(cached_hash) = self.file_cache.get(file_path) {
            if *cached_hash == hash {
                return Ok(vec![]);  // 文件未变更，跳过
            }
        }

        // 2. 检测语言
        let language = self.detect_language(file_path)?;

        // 3. Tree-sitter 解析
        let grammar = self.load_grammar(&language)?;
        let tree = grammar.parse(&content)?;
        
        // 4. 提取符号
        let symbols = self.extract_symbols(&tree, &content, file_path, &language);

        // 5. 向量嵌入 (如果启用)
        let entries: Vec<IndexEntry> = if let Some(embedder) = &self.embedder {
            let mut result = Vec::new();
            for sym in symbols {
                let embedding = embedder.embed(&sym.source_code).await?;
                result.push(IndexEntry {
                    id: uuid(),
                    file_path: file_path.to_string_lossy().into(),
                    symbol_name: sym.name,
                    symbol_kind: sym.kind,
                    start_line: sym.start_line,
                    end_line: sym.end_line,
                    source_code: sym.source_code,
                    embedding: Some(embedding),
                    language: language.clone(),
                    updated_at: current_timestamp(),
                });
            }
            result
        } else {
            symbols.into_iter().map(|sym| IndexEntry {
                id: uuid(),
                file_path: file_path.to_string_lossy().into(),
                symbol_name: sym.name,
                symbol_kind: sym.kind,
                start_line: sym.start_line,
                end_line: sym.end_line,
                source_code: sym.source_code,
                embedding: None,
                language: language.clone(),
                updated_at: current_timestamp(),
            }).collect()
        };

        // 6. 更新缓存
        // (file_cache 需要在 Indexer 中变为 RwLock 或使用 DashMap)
        
        Ok(entries)
    }

    /// 增量索引 (文件变更触发)
    pub async fn incremental_index(
        &mut self,
        changed_files: Vec<PathBuf>,
    ) -> Result<(), IndexError> {
        for file in &changed_files {
            // 删除旧索引
            self.db.delete_index_entries_for_file(
                &file.to_string_lossy()
            ).await?;
            
            // 重新索引
            let entries = self.index_file(file).await?;
            if !entries.is_empty() {
                self.db.batch_insert_index_entries(&entries).await?;
            }
        }
        Ok(())
    }

    /// 混合检索
    pub async fn search(
        &self,
        query: &str,
        top_k: usize,
    ) -> Result<Vec<SearchResult>, IndexError> {
        let mut all_results: Vec<(String, f64)> = Vec::new();

        // 1. FTS5 关键词检索
        let fts_results = self.db.fts5_search(query, top_k * 2).await?;
        all_results.extend(fts_results);

        // 2. 向量语义检索
        if let Some(embedder) = &self.embedder {
            let query_vec = embedder.embed(query).await?;
            let vec_results = self.db.vector_search(&query_vec, top_k * 2).await?;
            all_results.extend(vec_results);
        }

        // 3. RRF 融合排序
        let ranked = self.rrf_fusion(&all_results, top_k);

        // 4. 组装完整结果
        let mut results = Vec::new();
        for (entry_id, score) in ranked {
            if let Some(entry) = self.db.get_index_entry(&entry_id).await? {
                results.push(SearchResult {
                    entry,
                    score,
                    highlights: self.compute_highlights(&entry, query),
                });
            }
        }

        Ok(results)
    }

    /// RRF (Reciprocal Rank Fusion) 融合
    fn rrf_fusion(
        &self,
        results: &[(String, f64)],
        top_k: usize,
    ) -> Vec<(String, f64)> {
        let k: f64 = 60.0;  // RRF 常数
        
        // 按来源分组计算排名
        let mut scores: HashMap<String, f64> = HashMap::new();
        for (rank, (id, _)) in results.iter().enumerate() {
            let rrf_score = 1.0 / (k + (rank + 1) as f64);
            *scores.entry(id.clone()).or_insert(0.0) += rrf_score;
        }

        // 按融合分数排序
        let mut sorted: Vec<_> = scores.into_iter().collect();
        sorted.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(Ordering::Equal));
        sorted.truncate(top_k);
        
        sorted
    }
}
```

### 6.4 Rust 异步模式说明

> 本文档中大量使用 `#[async_trait]` 宏。Rust 原生 async trait (AFIT) 在 1.75 稳定，但仍有 Send/Sync 边界问题。

**使用 `#[async_trait]` 的原因**:
1. 兼容性: 支持更早的 Rust 版本
2. Send/Sync 明确性: 显式标注 `Send + Sync` 约束
3. 生态习惯: tokio/sqlx/tonic 等核心依赖都使用 async_trait

**注意事项**:
- `#[async_trait]` 会产生额外的 `Box<Pin<Future>>` 分配 (对性能影响极小)
- trait 对象 (`dyn Trait`) 必须包装在 `Arc` 中
- 避免在热路径 (每轮 PAOR 多次调用) 使用 trait 对象动态分发，改用泛型静态分发

### 6.5 内存管理与资源释放

`LoopEngineImpl` 持有多个 `Arc` 引用，需要明确定义资源清理顺序:

```rust
impl Drop for LoopEngineImpl {
    fn drop(&mut self) {
        // 1. 停止所有正在执行的 Sub-Agent
        // 2. 等待所有工具调用完成 (超时 5s)
        // 3. 保存最终检查点
        // 4. 关闭数据库连接
        // 5. 释放模型连接池
        log::info!("LoopEngine shut down gracefully");
    }
}

// 资源生命周期管理
// - 数据库连接: 由 ConnectionPool 管理，应用退出时统一关闭
// - 模型连接: 使用 reqwest Client 内置连接池，通过 Arc 共享
// - 向量嵌入模型: 加载后常驻内存，切换模型时释放旧模型
// - 文件句柄: 使用 RAII，确保不再使用时立即 drop
// - Tokio Runtime: 应用退出时调用 shutdown_timeout(30s) 等待任务完成
```
